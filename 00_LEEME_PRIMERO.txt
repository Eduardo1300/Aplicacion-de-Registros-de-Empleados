⚡ COMANDOS ÚTILES
═══════════════════════════════════════════════════════════════════════════════

# Iniciar todo
cd backend-nest && docker-compose up -d && cd ../frontend && npm run dev

# Ver logs
docker-compose logs -f

# Detener todo
docker-compose down

# Reset completo
docker-compose down -v && docker system prune -a -v

# Limpiar dependencias
rm -rf node_modules package-lock.json && npm install
