💡 EJEMPLOS RÁPIDOS
═══════════════════════════════════════════════════════════════════════════════

CREAR EMPLEADO:
───────────────────────────────────────────────────────────────────────────────
async saveEmpleado() {
  try {
    await api.createEmpleado(this.formData)
    this.notification.success('Empleado creado correctamente')
    this.loadEmpleados()
  } catch (err) {
    this.notification.error('Error al crear empleado')
  }
}


VALIDACIÓN:
───────────────────────────────────────────────────────────────────────────────
submitForm() {
  if (!this.name) {
    this.notification.warning('El nombre es obligatorio')
    return
  }
  this.save()
}


LOGIN:
───────────────────────────────────────────────────────────────────────────────
async login() {
  try {
    const res = await api.login(this.form)
    this.notification.success(`¡Bienvenido ${res.name}!`)
    this.$router.push('/dashboard')
  } catch (err) {
    this.notification.error('Credenciales inválidas')
  }
}


ELIMINAR:
───────────────────────────────────────────────────────────────────────────────
async deleteEmpleado(id) {
  try {
    await api.deleteEmpleado(id)
    this.notification.success('Eliminado correctamente')
    this.loadEmpleados()
  } catch (err) {
    this.notification.error('No se pudo eliminar')
  }
}
